cmake_minimum_required(VERSION 3.18)
project(PTCTLMC VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(PTCTLMC_ENABLE_CUDA  "Enable the CUDA GPU backend if a CUDA toolkit is found" ON)
option(PTCTLMC_ENABLE_METAL "Enable the Metal GPU backend on macOS if available" ON)
option(PTCTLMC_BUILD_TESTS  "Build tests" ON)

find_package(OpenMP QUIET)
# Apple Clang ships without an OpenMP runtime, so find_package fails even when
# Homebrew's libomp is installed. Wire it up by hand so the parallel network
# backend (--backend omp) is available out of the box on macOS. Non-Apple
# toolchains (GCC/Clang on Linux) find OpenMP natively above.
if(NOT OpenMP_CXX_FOUND AND APPLE)
  set(_omp_hints /opt/homebrew/opt/libomp /usr/local/opt/libomp)
  find_program(BREW_EXECUTABLE brew)
  if(BREW_EXECUTABLE)
    execute_process(COMMAND ${BREW_EXECUTABLE} --prefix libomp
                    OUTPUT_VARIABLE _omp_brew OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(_omp_brew)
      list(PREPEND _omp_hints "${_omp_brew}")
    endif()
  endif()
  foreach(_h ${_omp_hints})
    if(EXISTS "${_h}/lib/libomp.dylib")
      set(OpenMP_CXX_FLAGS "-Xpreprocessor -fopenmp -I${_h}/include" CACHE STRING "" FORCE)
      set(OpenMP_CXX_LIB_NAMES "omp" CACHE STRING "" FORCE)
      set(OpenMP_omp_LIBRARY "${_h}/lib/libomp.dylib" CACHE FILEPATH "" FORCE)
      find_package(OpenMP QUIET)
      break()
    endif()
  endforeach()
endif()
if(OpenMP_CXX_FOUND)
  message(STATUS "PTCTLMC: OpenMP ENABLED — parallel network backend (--backend omp)")
else()
  message(STATUS "PTCTLMC: OpenMP not found — network backend runs serially")
endif()

set(PTCTLMC_HAVE_CUDA OFF)
if(PTCTLMC_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    if(NOT CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES 70 80 90)
    endif()
    set(PTCTLMC_HAVE_CUDA ON)
    message(STATUS "PTCTLMC: CUDA backend ENABLED (${CMAKE_CUDA_COMPILER})")
  else()
    message(STATUS "PTCTLMC: no CUDA toolkit — building CPU-only")
  endif()
endif()

# Optional Metal backend (macOS only, auto-detected; independent of CUDA).
set(PTCTLMC_HAVE_METAL OFF)
if(PTCTLMC_ENABLE_METAL AND APPLE)
  find_library(PTCTLMC_METAL_FRAMEWORK      Metal)
  find_library(PTCTLMC_FOUNDATION_FRAMEWORK Foundation)
  if(PTCTLMC_METAL_FRAMEWORK AND PTCTLMC_FOUNDATION_FRAMEWORK)
    set(PTCTLMC_HAVE_METAL ON)
    message(STATUS "PTCTLMC: Metal backend ENABLED (${PTCTLMC_METAL_FRAMEWORK})")
  else()
    message(STATUS "PTCTLMC: Metal requested but frameworks not found — building without Metal")
  endif()
endif()

file(GLOB PTCTLMC_CORE_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(ptctlmc_core ${PTCTLMC_CORE_SRC})
target_include_directories(ptctlmc_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ptctlmc_core PUBLIC cxx_std_17)
if(OpenMP_CXX_FOUND)
  target_link_libraries(ptctlmc_core PUBLIC OpenMP::OpenMP_CXX)
  target_compile_definitions(ptctlmc_core PUBLIC PTCTLMC_HAVE_OPENMP=1)
endif()
if(PTCTLMC_HAVE_CUDA)
  file(GLOB PTCTLMC_CUDA_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/cuda/*.cu)
  if(PTCTLMC_CUDA_SRC)
    target_sources(ptctlmc_core PRIVATE ${PTCTLMC_CUDA_SRC})
    set_target_properties(ptctlmc_core PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
  endif()
  target_compile_definitions(ptctlmc_core PUBLIC PTCTLMC_HAVE_CUDA=1)
endif()
if(PTCTLMC_HAVE_METAL)
  file(GLOB PTCTLMC_METAL_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/metal/*.mm)
  if(PTCTLMC_METAL_SRC)
    target_sources(ptctlmc_core PRIVATE ${PTCTLMC_METAL_SRC})
    set_source_files_properties(${PTCTLMC_METAL_SRC} PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(ptctlmc_core PUBLIC
      ${PTCTLMC_METAL_FRAMEWORK} ${PTCTLMC_FOUNDATION_FRAMEWORK})
  endif()
  target_compile_definitions(ptctlmc_core PUBLIC PTCTLMC_HAVE_METAL=1)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ptctlmc_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wextra -Wno-unused-parameter>)
endif()

add_executable(ptctlmc ${CMAKE_CURRENT_SOURCE_DIR}/app/main.cpp)
target_link_libraries(ptctlmc PRIVATE ptctlmc_core)

if(PTCTLMC_BUILD_TESTS)
  enable_testing()
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt)
    add_subdirectory(tests)
  endif()
endif()
